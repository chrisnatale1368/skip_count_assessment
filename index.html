<script>

let currentAccuracy = null;

let currentFluency = null;

let attempts = JSON.parse(localStorage.getItem('skipCountAttempts')) || [];


function setAccuracy(value) {

  currentAccuracy = value;

  const labels = { correct: 'Accurate', selfcorrected: 'Self-corrected', error: 'Errors' };

  document.getElementById('accuracyDisplay').textContent = '→ ' + labels[value];

  updateCountCompletedVisibility();

}


function setFluency(value) {

  currentFluency = value;

  const labels = { fluent: 'Fluent throughout', moderate: 'Steady', hesitant: 'Hesitant/slow' };

  document.getElementById('fluencyDisplay').textContent = '→ ' + labels[value];

  updateCountCompletedVisibility();

}


function updateCountCompletedVisibility() {

  const countGroup = document.getElementById('countCompletedGroup');

  const countInput = document.getElementById('countCompleted');

  if ((currentAccuracy === 'error') || (currentFluency === 'hesitant')) {

    countGroup.style.display = 'flex';

    countInput.required = true;

  } else {

    countGroup.style.display = 'none';

    countInput.required = false;

    countInput.value = '';

  }

}


function saveAttempt() {

  const name = document.getElementById('studentName').value.trim();

  const interval = document.getElementById('interval').value;

  const start = document.getElementById('startPoint').value;

  const totalObjects = document.getElementById('totalObjects').value;

  const countCompleted = document.getElementById('countCompleted').value;

  const correspondence = document.getElementById('objectsCorrespondence').checked;

  const remainderStruggle = document.getElementById('remainderStruggle').checked;

  const notes = document.getElementById('notes').value.trim();

  

  if (!name) { alert('Please enter student name'); return; }

  if (!currentAccuracy || !currentFluency) { alert('Please select accuracy and fluency'); return; }

  if ((currentAccuracy === 'error' || currentFluency === 'hesitant') && !countCompleted) { alert('Please record how many counts were completed before the breakdown'); return; }

  

  const strategies = {

    movedGroups: document.getElementById('stratMovedGroups').checked,

    linedUp: document.getElementById('stratLinedUp').checked,

    separated: document.getElementById('stratSeparated').checked,

    containers: document.getElementById('stratContainers').checked,

    rows: document.getElementById('stratRows').checked,

    none: document.getElementById('stratNone').checked

  };

  

  const attempt = {

    id: Date.now(),

    timestamp: new Date().toLocaleString(),

    student: name,

    interval: interval,

    start: start,

    totalObjects: totalObjects || '—',

    countCompleted: countCompleted || '—',

    accuracy: currentAccuracy,

    fluency: currentFluency,

    correspondence: correspondence,

    remainderStruggle: remainderStruggle,

    strategies: strategies,

    notes: notes

  };

  

  attempts.unshift(attempt);

  localStorage.setItem('skipCountAttempts', JSON.stringify(attempts));

  

  document.getElementById('countCompleted').value = '';

  document.getElementById('notes').value = '';

  document.getElementById('objectsCorrespondence').checked = false;

  document.getElementById('remainderStruggle').checked = false;

  document.getElementById('stratMovedGroups').checked = false;

  document.getElementById('stratLinedUp').checked = false;

  document.getElementById('stratSeparated').checked = false;

  document.getElementById('stratContainers').checked = false;

  document.getElementById('stratRows').checked = false;

  document.getElementById('stratNone').checked = false;

  currentAccuracy = null;

  currentFluency = null;

  document.getElementById('accuracyDisplay').textContent = '';

  document.getElementById('fluencyDisplay').textContent = '';

  updateCountCompletedVisibility();

  renderAttempts();

}


function renderAttempts() {

  const list = document.getElementById('attemptsList');

  if (attempts.length === 0) { list.innerHTML = '<div class="empty-state">No attempts recorded yet</div>'; return; }

  list.innerHTML = attempts.map(att => {

    const accuracyLabel = { correct: 'Accurate', selfcorrected: 'Self-corrected', error: 'Errors' }[att.accuracy];

    const accuracyClass = att.accuracy === 'correct' ? 'success' : att.accuracy === 'selfcorrected' ? 'warning' : 'danger';

    const fluencyLabel = { fluent: 'Fluent', moderate: 'Steady', hesitant: 'Hesitant' }[att.fluency];

    const strategiesList = [];

    if (att.strategies.movedGroups) strategiesList.push('grouped objects');

    if (att.strategies.linedUp) strategiesList.push('lined up');

    if (att.strategies.separated) strategiesList.push('separated groups');

    if (att.strategies.containers) strategiesList.push('used containers');

    if (att.strategies.rows) strategiesList.push('visual pattern');

    if (att.strategies.none) strategiesList.push('no strategy');

    return `<div class="attempt-card"><div class="attempt-header"><div><div class="attempt-name">${att.student}</div><div class="attempt-timestamp">${att.timestamp}</div></div><button class="danger" onclick="deleteAttempt(${att.id})" style="padding: 6px 12px; font-size: 13px;">Delete</button></div><div class="attempt-meta">Skip by <strong>${att.interval}s</strong>, starting at <strong>${att.start}</strong>${att.totalObjects !== '—' ? ` • <strong>${att.totalObjects}</strong> objects` : ''}</div><div><span class="badge ${accuracyClass}">${accuracyLabel}</span><span class="badge neutral">${fluencyLabel}</span>${att.countCompleted !== '—' ? `<span class="badge neutral">${att.countCompleted} counts</span>` : ''}${att.correspondence ? '<span class="badge success">✓ Correspondence</span>' : ''}${att.remainderStruggle ? '<span class="badge warning">⚠ Remainder</span>' : ''}</div>${strategiesList.length > 0 ? `<div class="attempt-notes"><strong>Strategies:</strong> ${strategiesList.join(', ')}</div>` : ''}${att.notes ? `<div class="attempt-notes"><strong>Notes:</strong> "${att.notes}"</div>` : ''}</div>`;

  }).join('');

}


renderAttempts();


function buildSpreadsheetData() {

  const headers = ['Student Name', 'Assessment Date', 'Skip Interval', 'Starting Point', 'Total Objects', 'Counts Completed', 'Accuracy', 'Fluency', 'Objects Correspondence', 'Struggled with Remainder', 'Strategy: Moved Groups', 'Strategy: Lined Up', 'Strategy: Separated Groups', 'Strategy: Containers', 'Strategy: Rows/Pattern', 'Strategy: No Strategy', 'Notes'];

  const rows = attempts.map(att => [att.student, att.timestamp, 'By ' + att.interval + 's', att.start, att.totalObjects, att.countCompleted, att.accuracy === 'correct' ? 'Accurate' : att.accuracy === 'selfcorrected' ? 'Self-corrected' : 'Errors', att.fluency === 'fluent' ? 'Fluent' : att.fluency === 'moderate' ? 'Steady' : 'Hesitant', att.correspondence ? 'Yes' : 'No', att.remainderStruggle ? 'Yes' : 'No', att.strategies.movedGroups ? 'Yes' : 'No', att.strategies.linedUp ? 'Yes' : 'No', att.strategies.separated ? 'Yes' : 'No', att.strategies.containers ? 'Yes' : 'No', att.strategies.rows ? 'Yes' : 'No', att.strategies.none ? 'Yes' : 'No', att.notes]);

  return { headers, rows };

}


function downloadCSV() {

  if (attempts.length === 0) { alert('No attempts to export yet'); return; }

  const { headers, rows } = buildSpreadsheetData();

  let csv = headers.map(h => '"' + h.replace(/"/g, '""') + '"').join(',') + '\n';

  rows.forEach(row => { csv += row.map(cell => { const value = String(cell); return '"' + value.replace(/"/g, '""') + '"'; }).join(',') + '\n'; });

  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });

  const link = document.createElement('a');

  const url = URL.createObjectURL(blob);

  link.setAttribute('href', url);

  link.setAttribute('download', 'skip-count-assessments.csv');

  link.style.visibility = 'hidden';

  document.body.appendChild(link);

  link.click();

  document.body.removeChild(link);

}


function copyToClipboard() {

  if (attempts.length === 0) { alert('No attempts to copy yet'); return; }

  const { headers, rows } = buildSpreadsheetData();

  let text = headers.join('\t') + '\n';

  rows.forEach(row => { text += row.join('\t') + '\n'; });

  navigator.clipboard.writeText(text).then(() => { alert('Data copied to clipboard! Paste into Google Sheets or Excel.'); }).catch(() => { alert('Could not copy to clipboard'); });

}


function deleteAttempt(id) {

  if (confirm('Delete this attempt?')) {

    attempts = attempts.filter(a => a.id !== id);

    localStorage.setItem('skipCountAttempts', JSON.stringify(attempts));

    renderAttempts();

  }

}
